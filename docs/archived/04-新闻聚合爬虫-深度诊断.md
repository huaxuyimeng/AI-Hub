### 🔴 P0-1：RSS pubDate 解析失败时回退为 `new Date()` —— 用抓取时间冒充发布时间

**问题**：
1. **任意 `<a>` 都收** —— 抓到了导航、页脚、社交分享、"登录/注册"等链接
2. **长度阈值 10-200 太宽** —— "AI Bot"（7 字符）被丢但 "首页"（2 字符）也被丢（OK），但 "OpenAI 推出 GPT-5，多模态能力大幅提升" 这种短标题**完全可能 ≤ 10**

4. **没有过滤外链** —— 抓到的是站内导航"关于我们"完全可能 200 字符内

**修复**：

```ts
// src/lib/news/parsers/ai-bot.ts
function parseAiBot(html: string): FetchedItem[] {
  const items: FetchedItem[] = [];
  // ai-news-daily 同款正则：h2>a 标题 + p.text-muted 摘要 + 日期分组
  const dateMarkRe = /(\d{1,2})\s*月\s*(\d{1,2})\s*[·•]\s*周[一二三四五六日]/g;
  const dateMarks: { idx: number; ts: number }[] = [];
  let dm;
  while ((dm = dateMarkRe.exec(html)) !== null) {
    const month = parseInt(dm[1], 10);
    const day = parseInt(dm[2], 10);
    const now = new Date();
    let year = now.getFullYear();
    const candidate = new Date(year, month - 1, day);
    // 跨年回退
    if (candidate.getTime() - now.getTime() > 86400000) {
      candidate.setFullYear(year - 1);
    }
    dateMarks.push({ idx: dm.index, ts: candidate.getTime() });
  }

  const itemRe = /<h2[^>]*>\s*<a[^>]+href="(https?:\/\/[^"]+)"[^>]*>([\s\S]*?)<\/a>\s*<\/h2>/g;
  let m;
  while ((m = itemRe.exec(html)) !== null) {
    const url = m[1];
    const title = cleanText(m[2]);
    if (!title || title.length < 4) continue;

    const after = html.slice(m.index + m[0].length, m.index + m[0].length + 2500);
    const pM = after.match(/<p[^>]*class="text-muted text-sm"[^>]*>([\s\S]*?)<\/p>/);
    let summary = '';
    if (pM) {
      summary = pM[1]
        .replace(/<span class="news-time[\s\S]*?<\/span>/g, '')
        .replace(/<a[^>]*>([\s\S]*?)<\/a>/g, '$1');
      summary = cleanText(summary).slice(0, 300);
    }

    // 真实发布日期
    let publishedAt: number | null = null;
    let precision: 'exact' | 'date' | null = null;
    for (const mk of dateMarks) {
      if (mk.idx <= m.index) {
        publishedAt = mk.ts;
        precision = 'date';
      }
    }

    items.push({
      id: hashId(url + title),
      title, summary, url,
      source: 'ai-bot',
      sourceLabel: 'AI Bot 每日资讯',
      publishedAt: publishedAt ? new Date(publishedAt) : null,
      publishPrecision: precision,
      crawledAt: new Date(),
    });
  }
  return items;
}
```

**问题**：
- README §"快速开始"提到"查询所有用户的刷新频率，只处理到达时间窗口的用户"
- **实际 cron 直接调用 `fetchAllNews()`，完全不读 UserPreferences**
- 后果：所有用户都被同等对待，频率设置是死字段

**修复方案**：

```ts
// src/app/api/cron/fetch-news/route.ts
export async function GET(req: Request) {
  const authHeader = req.headers.get('authorization');
  const expected = process.env.CRON_SECRET;
  if (expected && authHeader !== `Bearer ${expected}`) {
    return NextResponse.json({ error: 'Unauthorized' }, { status: 401 });
  }

  const startTime = Date.now();
  const url = new URL(req.url);
  const onlyUserId = url.searchParams.get('userId');  // 支持单用户手动触发

  try {
    // 找出本轮该刷新的用户
    const now = new Date();
    const users = await prisma.userPreferences.findMany({
      where: {
        ...(onlyUserId ? { userId: onlyUserId } : {}),
      },
      select: {
        userId: true,
        tenantId: true,
        newsRefreshInterval: true,
        newsCategories: true,
        newsSources: true,
        followedModels: true,
      },
    });

    // 按频率分组
    const dueUsers: typeof users = []
    for (const u of users) {
      // 检查上次刷新时间
      const lastFetch = await prisma.usageStat.findFirst({
        where: { tenantId: u.tenantId /* 用某个 NewsFetchLog 表替代 */ },
        orderBy: { date: 'desc' },
      })
      const hoursSince = lastFetch ? (now.getTime() - lastFetch.date.getTime()) / 3600000 : Infinity
      if (hoursSince >= u.newsRefreshInterval) {
        dueUsers.push(u)
      }
    }

    // 按用户偏好抓取
    let totalItems = 0
    const sources: Array<{ name: string; count: number; success: boolean }> = []
    const warnings: string[] = []
    const errors: string[] = []

    for (const u of dueUsers) {
      const userCategories = u.newsCategories ? u.newsCategories.split(',').filter(Boolean) : []
      const userSources = u.newsSources ? u.newsSources.split(',').filter(Boolean) : []
      const userModels = u.followedModels ? u.followedModels.split(',').filter(Boolean) : []

      const result = await fetchAllNews({
        categories: userCategories,
        sources: userSources,
        followedModels: userModels,
      })
      totalItems += result.totalItems
      sources.push(...result.sources)
      warnings.push(...(result.warnings ?? []))
      errors.push(...(result.errors ?? []))
    }

    return NextResponse.json({
      ok: true,
      duration: Date.now() - startTime,
      userCount: dueUsers.length,
      totalItems,
      sources,
      warnings,
      errors,
      timestamp: new Date().toISOString(),
    });
  } catch (err) {
    console.error('[cron] News fetch failed:', err);
    return NextResponse.json({ error: 'Internal error', code: 'NEWS_FETCH_FAILED' }, { status: 500 });
  }
}
```

```ts
// src/lib/news/entity-fingerprint.ts  （已在上文给出完整代码）

// src/lib/news/service.ts §dedupeAndCrossV2
function dedupeAndCrossV2(items: FetchedItem[]): FetchedItem[] {
  // 第一步：粗合并（精确标题 / Jaccard ≥ 0.7，避免误合）
  const groups: FetchedItem[][] = []
  for (const item of items) {
    let matched = false
    for (const group of groups) {
      if (item.title === group[0].title || titleSimilarity(item.title, group[0].title) >= 0.7) {
        group.push(item)
        matched = true
        break
      }
    }
    if (!matched) groups.push([item])
  }

  // 第二步：实体指纹跨组多源关联
  const flat = groups.flat().map((g) => ({
    id: g.url,
    source: g.sourceName,
    title: g.title,
    summary: g.summary ?? null,
    ref: g,
  }))
  const relations = findMultiSourceRelations(flat)

  // 写回 crossSources
  const byUrl = new Map<string, FetchedItem>()
  for (const grp of groups) byUrl.set(grp[0].url, grp[0])
  for (const rel of relations) {
    const item = byUrl.get(rel.itemId)
    if (item) {
      item.crossSources = Array.from(new Set([...(item.crossSources ?? []), ...rel.sources.filter(s => s !== item.sourceName)]))
    }
  }

  return groups.map((g) => {
    g.sort((a, b) => {
      const aT = a.publishedAt?.getTime() ?? Number.MAX_SAFE_INTEGER
      const bT = b.publishedAt?.getTime() ?? Number.MAX_SAFE_INTEGER
      return aT - bT
    })
    const primary = g[0]
    const otherSources = g.slice(1).map((x) => x.sourceName).filter(Boolean)
    return {
      ...primary,
      crossSources: Array.from(new Set([...(primary.crossSources ?? []), ...otherSources])),
    }
  })
}
```

**问题**：
- 5 个一组**串行等待** —— 实际就是「分批并发」，总时间 = ceil(N/5) × 最慢源
- 没有**全局并发数限制** —— 20 个源会一次全开（如果代码改成 `Promise.all`）
- 没有**每个源独立超时** —— 单源卡死会拖累整批
- 没有**重试** —— 瞬时抖动就丢整天数据

**修复**：用 p-limit 或自实现令牌桶

```ts
// src/lib/utils/concurrency.ts
export class Semaphore {
  private available: number
  private waiting: Array<() => void> = []
  constructor(concurrency: number) { this.available = concurrency }
  async acquire(): Promise<void> {
    if (this.available > 0) {
      this.available--
      return
    }
    return new Promise<void>(resolve => this.waiting.push(resolve))
  }
  release(): void {
    const next = this.waiting.shift()
    if (next) next()
    else this.available++
  }
}

// src/lib/news/service.ts §fetchAllNews
const sem = new Semaphore(5)
const results = await Promise.allSettled(
  configs.map(async (cfg) => {
    await sem.acquire()
    try {
      const t0 = Date.now()
      const items = await fetchFromSourceWithRetry(cfg, 2)  // 重试 2 次
      return { name: cfg.name, items, ms: Date.now() - t0 }
    } finally {
      sem.release()
    }
  })
)

async function fetchFromSourceWithRetry(cfg: NewsSourceConfig, maxRetries = 2): Promise<FetchedItem[]> {
  for (let attempt = 0; attempt <= maxRetries; attempt++) {
    try {
      return await fetchFromSource(cfg)
    } catch (e) {
      if (attempt === maxRetries) throw e
      await new Promise(r => setTimeout(r, 1000 * (attempt + 1)))  // 1s, 2s 退避
    }
  }
  throw new Error('unreachable')
}
```

- 一轮抓取 50-100 条，单源再多就有问题

```ts
async function saveItems(items: FetchedItem[]): Promise<number> {
  let saved = 0

  // 第一步：批量获取/创建 source
  const sourceNames = Array.from(new Set(items.map(it => it.sourceName).filter(Boolean)))
  const existingSources = await prisma.newsSource.findMany({
    where: { name: { in: sourceNames } },
  })
  const sourceMap = new Map(existingSources.map(s => [s.name, s]))

  // 创建缺失的 source
  const missing = sourceNames.filter(n => !sourceMap.has(n))
  if (missing.length > 0) {
    await prisma.newsSource.createMany({
      data: missing.map(name => ({
        name,
        url: '',
        type: 'rss',
        lastFetchAt: new Date(),
      })),
    })
    // 重新查
    const newSources = await prisma.newsSource.findMany({
      where: { name: { in: missing } },
    })
    newSources.forEach(s => sourceMap.set(s.name, s))
  }

  // 第二步：批量 upsert newsItem
  // 注意：Prisma 不支持真正的 batch upsert，需要分块用事务
  const BATCH_SIZE = 50
  for (let i = 0; i < items.length; i += BATCH_SIZE) {
    const batch = items.slice(i, i + BATCH_SIZE)
    try {
      await prisma.$transaction(
        batch.map(item => {
          const source = sourceMap.get(item.sourceName)
          if (!source) throw new Error(`Source not found: ${item.sourceName}`)
          const relatedModels = extractModelNames(`${item.title} ${item.summary ?? ''}`).join(',')
          const crossSources = (item.crossSources ?? []).join(',')
          const category = classifyCategory(item.title, item.summary)
          return prisma.newsItem.upsert({
            where: { url: item.url },
            update: {
              summary: item.summary,
              relatedModels,
              crossSources,
              category: category ?? undefined,
              publishPrecision: item.publishPrecision ?? undefined,
            },
            create: {
              title: item.title,
              url: item.url,
              sourceId: source.id,
              summary: item.summary,
              content: item.content,
              publishedAt: item.publishedAt ?? new Date(),  // ⚠️ 仍有 P0-1 风险
              publishPrecision: item.publishPrecision,
              category: category ?? undefined,
              relatedModels,
              crossSources,
              confidence: crossSources.length > 0 ? 'A' : 'B',
            },
          })
        })
      )
      saved += batch.length
    } catch (err) {
      console.error(`[news] Batch save failed:`, (err as Error).message)
    }
  }
  return saved
}
```

**注意**：批量时 `publishedAt ?? new Date()` 仍有 P0-1 风险，应该直接用 `publishedAt: item.publishedAt ?? new Date(0)` 让 null 表现为 1970-01-01，比用抓取时间更安全。或加 schema 让字段可空：

```prisma
model NewsItem {
  ...
  publishedAt DateTime?  // ← 可空
  publishPrecision String?  // 'exact' | 'date' | null
  ...
}
```

```ts
export async function getAvailableDates(): Promise<string[]> {
```

```ts
    WHERE deletedAt IS NULL
    ORDER BY date DESC
    LIMIT 365
  `
  return result.map(r => r.date)
}
```

```ts
const redis = createClient({
  url: process.env.UPSTASH_REDIS_REST_URL!,
  token: process.env.UPSTASH_REDIS_REST_TOKEN!,
})

export async function withLock<T>(
  key: string,
  ttlSeconds: number,
  fn: () => Promise<T>
): Promise<T | { skipped: true; reason: string }> {
  // 原子获取锁
  const acquired = await redis.set(key, '1', { nx: true, ex: ttlSeconds })
  if (acquired !== 'OK') {
    return { skipped: true, reason: 'lock held' }
  }
  try {
    return await fn()
  } finally {
    await redis.del(key)
  }
}
```

```ts
async function fetchHtml(url: string, sourceName: string): Promise<FetchedItem[]> {
  try {
    const res = await fetch(url, {
      headers: { 'User-Agent': 'Mozilla/5.0 (compatible; AIHub-Bot)' },
      signal: AbortSignal.timeout(20_000),
    });
    if (!res.ok) return [];

    const html = await res.text();
    const items: FetchedItem[] = [];
    const baseUrl = new URL(url);
    const linkRegex = /<a[^>]+href="([^"]+)"[^>]*>([\s\S]*?)<\/a>/gi;
```
