## 🔧 技术实现

### 核心代码

```typescript
class FileLock {
  async acquire(timeoutMs = 0): Promise<boolean> {
    while (true) {
      try {
        // 原子操作：文件存在则失败
        fs.writeFileSync(lockPath, JSON.stringify({
          pid: process.pid,
          acquiredAt: new Date().toISOString(),
        }), { flag: 'wx' });
        
        return true; // 获取成功
      } catch (err) {
        if (err.code === 'EEXIST') {
          // 检查僵尸锁
          const age = Date.now() - stat.mtimeMs;
          if (age > 10 * 60 * 1000) {
            fs.unlinkSync(lockPath); // 清理
            continue; // 重试
          }
          
          // 超时或不等待
          if (timeoutMs === 0) return false;
          
          // 等待重试
          await sleep(100);
        }
      }
    }
  }
}
```

### 便捷函数

```typescript
export async function withLock<T>(
  key: string,
  ttlSeconds: number,
  fn: () => Promise<T>,
): Promise<T | { skipped: true; reason: string }> {
  const lock = new FileLock(key);
  const acquired = await lock.acquire(0);

  if (!acquired) {
    return { skipped: true, reason: 'lock held (file-based)' };
  }

  try {
    return await fn();
  } finally {
    lock.release();
  }
}
```

```typescript
import Redis from 'ioredis';

export async function withRedisLock<T>(
  redis: Redis,
  key: string,
  ttlSeconds: number,
  fn: () => Promise<T>
): Promise<T | null> {
  const lockKey = `lock:${key}`;
  const lockValue = randomUUID();
  
  // SET key value NX EX ttl
  const acquired = await redis.set(
    lockKey, 
    lockValue, 
    'EX', 
    ttlSeconds, 
    'NX'
  );
  
  if (!acquired) return null;
  
  try {
    return await fn();
  } finally {
    // Lua 脚本原子释放
    await redis.eval(
      `if redis.call("get",KEYS[1]) == ARGV[1] then
         return redis.call("del",KEYS[1])
       else
         return 0
       end`,
      1,
      lockKey,
      lockValue
    );
  }
}
```
