```ts
export type LogLevel = 'debug' | 'info' | 'warn' | 'error'

export interface LogContext {
  cron?: string
  userId?: string
  tenantId?: string
  source?: string
  modelId?: string
  duration?: number
  count?: number
  warningCount?: number
  errorCount?: number
  error?: string
  stack?: string
  [key: string]: unknown
}

export function log(level: LogLevel, message: string, ctx: LogContext = {}) {
  const entry = {
    timestamp: new Date().toISOString(),
    level,
    message,
    ...ctx,
  }
  // 输出 JSON 行，方便 grep / 日志聚合（Datadog/Loki/CloudWatch）
  const line = JSON.stringify(entry)
  if (level === 'error') console.error(line)
  else if (level === 'warn') console.warn(line)
  else console.log(line)
}

export const logger = {
  debug: (msg: string, ctx?: LogContext) => log('debug', msg, ctx),
  info: (msg: string, ctx?: LogContext) => log('info', msg, ctx),
  warn: (msg: string, ctx?: LogContext) => log('warn', msg, ctx),
  error: (msg: string, ctx?: LogContext) => log('error', msg, ctx),
}
```

// src/app/api/cron/fetch-news/route.ts
import { logger } from '@/lib/observability/logger'
import { withLock } from '@/lib/observability/distributed-lock'

export async function GET(req: Request) {
  const authHeader = req.headers.get('authorization')
  const expected = process.env.CRON_SECRET
  if (expected && authHeader !== `Bearer ${expected}`) {
    return NextResponse.json({ error: 'Unauthorized' }, { status: 401 })
  }

  return withLock('cron:fetch-news', 600, async () => {
    const start = Date.now()
    const cronName = 'fetch-news'
    try {
      const result = await fetchAllNews()
      logger.info('cron completed', {
        cron: cronName,
        duration: Date.now() - start,
        totalItems: result.totalItems,
        sourceCount: result.sources.length,
        successCount: result.sources.filter(s => s.success).length,
        warningCount: result.warnings?.length ?? 0,
        errorCount: result.errors?.length ?? 0,
      })
      return NextResponse.json({
        ok: true,
        duration: Date.now() - start,
        totalItems: result.totalItems,
        sources: result.sources.map(s => ({
          name: s.name, count: s.count, success: s.success,
        })),
        warnings: result.warnings,
        errors: result.errors,
      })
    } catch (err) {
      logger.error('cron failed', {
        cron: cronName,
        duration: Date.now() - start,
        error: (err as Error).message,
        stack: (err as Error).stack,
      })
      // 客户端不回传内部错误
      return NextResponse.json({
        error: 'Internal error',
        code: 'NEWS_FETCH_FAILED',
      }, { status: 500 })
    }
  })
}

### 3.2 分布式锁（S-2 / S-3）

**新增文件**：`src/lib/observability/distributed-lock.ts`

```ts
import { Redis } from '@upstash/redis'

const redis = process.env.UPSTASH_REDIS_REST_URL
  ? new Redis({
      url: process.env.UPSTASH_REDIS_REST_URL,
      token: process.env.UPSTASH_REDIS_REST_TOKEN!,
    })
  : null

/**
 * 分布式锁。Vercel 多实例时保证同一 cron 不并发跑。
 * 无 Redis 时退化为进程内锁（Vercel 单实例足够）。
 */
export async function withLock<T>(
  key: string,
  ttlSeconds: number,
  fn: () => Promise<T>
): Promise<T | { skipped: true; reason: string }> {
  if (redis) {
    const acquired = await redis.set(key, process.pid.toString(), { nx: true, ex: ttlSeconds })
    if (acquired !== 'OK') {
      logger.warn('lock contended', { key })
      return { skipped: true, reason: 'lock held' }
    }
    try {
      return await fn()
    } finally {
      await redis.del(key)
    }
  }

  // 进程内锁（开发环境 / 无 Redis）
  const inProc = inProcLocks.get(key)
  if (inProc && Date.now() < inProc) {
    return { skipped: true, reason: 'lock held (in-process)' }
  }
  inProcLocks.set(key, Date.now() + ttlSeconds * 1000)
  try {
    return await fn()
  } finally {
    inProcLocks.delete(key)
  }
}

const inProcLocks = new Map<string, number>()
```

### 3.3 重试 + 退避 + 单源超时（S-4 / S-5 / S-6）

**新增文件**：`src/lib/utils/fetch-with-retry.ts`

```ts
export interface FetchRetryOptions {
  maxRetries?: number
  timeoutMs?: number
  backoffMs?: number
  retryable?: (err: Error) => boolean
}

export async function fetchWithRetry(
  url: string,
  init: RequestInit = {},
  options: FetchRetryOptions = {}
): Promise<Response> {
  const {
    maxRetries = 2,
    timeoutMs = 15_000,
    backoffMs = 1500,
    retryable = () => true,
  } = options

  let lastError: Error | null = null

  for (let attempt = 0; attempt <= maxRetries; attempt++) {
    const ctl = new AbortController()
    const timer = setTimeout(() => ctl.abort(new Error('timeout')), timeoutMs)

    try {
      const res = await fetch(url, {
        ...init,
        signal: ctl.signal,
      })
      clearTimeout(timer)
      if (res.ok) return res
      if (res.status >= 500 || res.status === 429) {
        // 服务端错误 / 限流 → 重试
        lastError = new Error(`HTTP ${res.status}`)
      } else {
        // 客户端错误（4xx 但不是 429）→ 不重试
        return res
      }
    } catch (e) {
      clearTimeout(timer)
      lastError = e as Error
      if (!retryable(lastError)) throw lastError
    }

    if (attempt < maxRetries) {
      await new Promise(r => setTimeout(r, backoffMs * (attempt + 1)))
    }
  }

  throw lastError ?? new Error('fetchWithRetry: max retries reached')
}
```

// src/lib/rankings/scraper.ts §refreshAll
async refreshAll(): Promise<{ total: number; succeeded: number; failed: number; details: any[] }> {
  const models = await prisma.model.findMany({ where: { isActive: true } })
  const total = models.length

  // 创建 CronJob
  const job = await prisma.cronJob.create({
    data: { name: 'refresh-models', triggeredBy: 'cron' },
  })

  // 进度：每 5 个更新一次
  const UPDATE_INTERVAL = 5
  let completed = 0
  const details: any[] = []

  // 分批并发（每批 5 个）
  for (let i = 0; i < models.length; i += 5) {
    const batch = models.slice(i, i + 5)
    const batchResults = await Promise.allSettled(
      batch.map(async m => {
        try {
          const result = await this.refreshOne(m.id)
          return { id: m.id, status: 'ok' as const }
        } catch (e) {
          return { id: m.id, status: 'failed' as const, error: (e as Error).message }
        }
      })
    )
    for (const r of batchResults) details.push(r.status === 'fulfilled' ? r.value : { status: 'failed' })

    completed += batch.length
    if (completed % UPDATE_INTERVAL === 0 || completed === total) {
      await prisma.cronJob.update({
        where: { id: job.id },
        data: { progress: { completed, total, details: details.slice(-10) } as any },
      })
    }
  }

  // 完成
  await prisma.cronJob.update({
    where: { id: job.id },
    data: {
      finishedAt: new Date(),
      status: details.some(d => d.status === 'failed') ? 'succeeded' : 'succeeded',  // 即使有部分失败也标 succeeded
      progress: { completed, total, details } as any,
    },
  })

  return {
    total,
    succeeded: details.filter(d => d.status === 'ok').length,
    failed: details.filter(d => d.status === 'failed').length,
    details,
  }
}
