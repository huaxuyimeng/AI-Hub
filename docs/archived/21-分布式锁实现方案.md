# 分布式锁实现方案

**目标**：防止 `/api/cron/fetch-news` 并发执行  
**场景**：多个 cron 任务或手动触发同时调用

---

## 方案对比

### 方案 A：文件锁（推荐）✅

**优点**：
- 无需额外依赖
- 实现简单
- 跨进程有效

**缺点**：
- 不适合多服务器部署
- 文件系统依赖

**适用场景**：单服务器部署（当前场景）

---

### 方案 B：Redis 锁

**优点**：
- 支持分布式
- 性能好
- 可设置过期时间

**缺点**：
- 需要 Redis 服务
- 增加依赖复杂度

**适用场景**：多服务器部署

---

### 方案 C：数据库锁

**优点**：
- 利用现有 Prisma
- 分布式支持

**缺点**：
- 增加 DB 负载
- 实现复杂（需要清理过期锁）

**适用场景**：无文件系统或 Redis 时

---

## 推荐实现：文件锁

### 实现代码

```typescript
// src/lib/utils/file-lock.ts

import fs from 'fs';
import path from 'path';
import os from 'os';

export class FileLock {
  private lockPath: string;
  private acquired = false;

  constructor(lockName: string) {
    const tmpDir = os.tmpdir();
    this.lockPath = path.join(tmpDir, `${lockName}.lock`);
  }

  /**
   * 尝试获取锁
   * @param timeoutMs 超时时间（毫秒），0 表示不等待
   * @returns 是否成功获取锁
   */
  async acquire(timeoutMs = 0): Promise<boolean> {
    const startTime = Date.now();

    while (true) {
      try {
        // 使用 wx 模式：文件存在时抛出错误
        fs.writeFileSync(this.lockPath, JSON.stringify({
          pid: process.pid,
          acquiredAt: new Date().toISOString(),
        }), { flag: 'wx' });

        this.acquired = true;
        return true;
      } catch (err: any) {
        // 文件已存在（锁被占用）
        if (err.code === 'EEXIST') {
          // 检查锁是否过期（超过 10 分钟自动清理）
          try {
            const stat = fs.statSync(this.lockPath);
            const age = Date.now() - stat.mtimeMs;
            if (age > 10 * 60 * 1000) {
              // 锁文件超过 10 分钟，认为是僵尸锁，强制删除
              fs.unlinkSync(this.lockPath);
              continue; // 重试获取
            }
          } catch {
            // stat 失败，可能已被删除，重试
            continue;
          }

          // 超时检查
          if (timeoutMs === 0) {
            return false;
          }
          if (Date.now() - startTime >= timeoutMs) {
            return false;
          }

          // 等待 100ms 后重试
          await new Promise(resolve => setTimeout(resolve, 100));
        } else {
          // 其他错误（权限、磁盘满等）
          throw err;
        }
      }
    }
  }

  /**
   * 释放锁
   */
  release(): void {
    if (!this.acquired) return;

    try {
      fs.unlinkSync(this.lockPath);
      this.acquired = false;
    } catch (err: any) {
      if (err.code !== 'ENOENT') {
        // 文件不存在是正常的（可能被其他进程清理）
        console.error('Failed to release lock:', err);
      }
    }
  }

  /**
   * 检查锁是否被占用
   */
  isLocked(): boolean {
    try {
      fs.accessSync(this.lockPath, fs.constants.F_OK);
      return true;
    } catch {
      return false;
    }
  }
}

/**
 * 便捷函数：使用锁执行函数
 */
export async function withLock<T>(
  lockName: string,
  fn: () => Promise<T>,
  options?: { timeout?: number }
): Promise<T | null> {
  const lock = new FileLock(lockName);
  const acquired = await lock.acquire(options?.timeout ?? 0);

  if (!acquired) {
    return null; // 获取锁失败
  }

  try {
    return await fn();
  } finally {
    lock.release();
  }
}
```

## 使用方式

### 方式 1：手动控制

```typescript
import { FileLock } from '@/lib/utils/file-lock';

export async function POST(req: NextRequest) {
  const lock = new FileLock('fetch-news');
  const acquired = await lock.acquire(0); // 不等待，立即返回

  if (!acquired) {
    return NextResponse.json(
      { ok: false, error: 'Another fetch is in progress' },
      { status: 409 }
    );
  }

  try {
    // 执行抓取逻辑
    const result = await fetchAllNews();
    return NextResponse.json(result);
  } finally {
    lock.release();
  }
}
```

### 方式 2：便捷函数

```typescript
import { withLock } from '@/lib/utils/file-lock';

export async function POST(req: NextRequest) {
  const result = await withLock('fetch-news', async () => {
    return await fetchAllNews();
  });

  if (result === null) {
    return NextResponse.json(
      { ok: false, error: 'Another fetch is in progress' },
      { status: 409 }
    );
  }

  return NextResponse.json(result);
}
```

## 测试用例

```typescript
// tests/file-lock.test.ts

import { FileLock, withLock } from '@/lib/utils/file-lock';

describe('FileLock', () => {
  it('should acquire and release lock', async () => {
    const lock = new FileLock('test-lock');
    
    const acquired = await lock.acquire();
    expect(acquired).toBe(true);
    expect(lock.isLocked()).toBe(true);
    
    lock.release();
    expect(lock.isLocked()).toBe(false);
  });

  it('should prevent concurrent access', async () => {
    const lock1 = new FileLock('concurrent-test');
    const lock2 = new FileLock('concurrent-test');
    
    await lock1.acquire();
    
    const acquired2 = await lock2.acquire(0);
    expect(acquired2).toBe(false);
    
    lock1.release();
    
    const acquired2Retry = await lock2.acquire(0);
    expect(acquired2Retry).toBe(true);
    
    lock2.release();
  });

  it('should auto-cleanup stale locks', async () => {
    const lock = new FileLock('stale-test');
    
    // 模拟创建 11 分钟前的锁文件
    // (实际测试中需要 mock fs.statSync)
    
    const acquired = await lock.acquire();
    expect(acquired).toBe(true);
    
    lock.release();
  });

  it('withLock should handle errors', async () => {
    const result = await withLock('error-test', async () => {
      throw new Error('Test error');
    });
    
    // 即使出错，锁也应该被释放
    const lock = new FileLock('error-test');
    expect(lock.isLocked()).toBe(false);
  });
});
```

## 集成到 cron API

```typescript
// src/app/api/cron/fetch-news/route.ts

import { NextRequest, NextResponse } from 'next/server';
import { withLock } from '@/lib/utils/file-lock';
import { fetchAllNews } from '@/lib/news/service';

export async function POST(req: NextRequest) {
  // 验证 Authorization
  const auth = req.headers.get('authorization');
  if (auth !== `Bearer ${process.env.CRON_SECRET}`) {
    return NextResponse.json({ error: 'Unauthorized' }, { status: 401 });
  }

  // 使用锁执行抓取
  const result = await withLock('fetch-news', async () => {
    return await fetchAllNews();
  });

  // 锁被占用（另一个抓取正在进行）
  if (result === null) {
    return NextResponse.json(
      {
        ok: false,
        error: 'Another fetch is in progress',
        message: '已有抓取任务正在执行，请稍后重试'
      },
      { status: 409 } // 409 Conflict
    );
  }

  // 抓取成功
  return NextResponse.json(result);
}
```

## 监控和日志

```typescript
// src/lib/utils/file-lock.ts (扩展)

import { logInfo, logError } from '@/lib/observability/logger';

export class FileLock {
  // ... 现有代码 ...

  async acquire(timeoutMs = 0): Promise<boolean> {
    const startTime = Date.now();
    logInfo('lock.acquire.attempt', {
      lockName: this.lockName,
      timeout: timeoutMs
    });

    while (true) {
      try {
        fs.writeFileSync(this.lockPath, JSON.stringify({
          pid: process.pid,
          acquiredAt: new Date().toISOString(),
        }), { flag: 'wx' });

        this.acquired = true;
        logInfo('lock.acquire.success', {
          lockName: this.lockName,
          duration: Date.now() - startTime
        });
        return true;
      } catch (err: any) {
        if (err.code === 'EEXIST') {
          // 检查僵尸锁...
          try {
            const stat = fs.statSync(this.lockPath);
            const age = Date.now() - stat.mtimeMs;
            if (age > 10 * 60 * 1000) {
              logInfo('lock.cleanup.stale', {
                lockName: this.lockName,
                age: age
              });
              fs.unlinkSync(this.lockPath);
              continue;
            }
          } catch {
            continue;
          }

          // 超时或不等待
          if (timeoutMs === 0 || Date.now() - startTime >= timeoutMs) {
            logInfo('lock.acquire.failed', {
              lockName: this.lockName,
              reason: timeoutMs === 0 ? 'no-wait' : 'timeout'
            });
            return false;
          }

          await new Promise(resolve => setTimeout(resolve, 100));
        } else {
          logError('lock.acquire.error', err);
          throw err;
        }
      }
    }
  }

  release(): void {
    if (!this.acquired) return;

    try {
      fs.unlinkSync(this.lockPath);
      this.acquired = false;
      logInfo('lock.release.success', { lockName: this.lockName });
    } catch (err: any) {
      if (err.code !== 'ENOENT') {
        logError('lock.release.error', err);
      }
    }
  }
}
```
